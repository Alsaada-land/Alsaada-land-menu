        window.initDatabase = async function() {
            if (!confirm("هل أنت متأكد من رغبتك في رفع كافة أصناف المنيو؟")) return;
            try {
                const collectionRef = collection(db, "menu");
                for (let item of initialMenuItems) {
                    await addDoc(collectionRef, item);
                }
                alert("تم رفع كافة أصناف المنيو بنجاح!");
                fetchAdminMenu();
            } catch (error) {
                alert("حدث خطأ أثناء الرفع");
                console.error(error);
            }
        }

        async function fetchAdminMenu() {
            const listDiv = document.getElementById('adminList');
            listDiv.innerHTML = 'جاري التحميل...';
            
            const querySnapshot = await getDocs(collection(db, "menu"));
            listDiv.innerHTML = '';
            
            if (querySnapshot.empty) {
                listDiv.innerHTML = '<div style="text-align:center; color:var(--text-muted); padding:15px;">قاعدة البيانات فارغة. اضغط على الزر الأخضر بالأسفل لرفع الأصناف!</div>';
                return;
            }

            querySnapshot.forEach((document) => {
                const item = document.data();
                const id = document.id;

                const row = document.createElement('div');
                row.className = 'item-row';
                row.innerHTML = `
                    <div><strong>${item.name}</strong> <span style="font-size:0.8rem; color:var(--text-muted);">(${item.cat})</span></div>
                    <div class="item-info">
                        <input type="number" id="price_${id}" value="${item.price}">
                        <button class="save-price-btn" onclick="updatePrice('${id}')">حفظ</button>
                    </div>
                `;
                listDiv.appendChild(row);
            });
        }

        window.updatePrice = async function(id) {
            const newPrice = document.getElementById(`price_${id}`).value;
            try {
                const itemRef = doc(db, "menu", id);
                await updateDoc(itemRef, { price: Number(newPrice) });
                alert("تم تحديث السعر بنجاح!");
            } catch (error) {
                alert("حدث خطأ أثناء التحديث");
            }
        }

        window.addNewItem = async function() {
            const name = document.getElementById('newName').value;
            const price = document.getElementById('newPrice').value;
            const cat = document.getElementById('newCat').value;
            const desc = document.getElementById('newDesc').value;
            const img = document.getElementById('newImg').value || "https://images.unsplash.com/photo-1555939594-58d7cb561ad1?w=500";

            if (!name || !price) {
                alert("الرجاء إدخال الاسم والسعر!");
                return;
            }

            try {
                await addDoc(collection(db, "menu"), { name, price: Number(price), cat, desc, img });
                alert("تمت الإضافة بنجاح!");
                fetchAdminMenu();
            } catch (error) {
                alert("خطأ أثناء الإضافة");
            }
        }

        fetchAdminMenu();
